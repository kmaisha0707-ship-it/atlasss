// 2. ПРЕПАРАТТАР ТІЗІМІ

let currentIndex = 0;

function renderList() {

    const list = document.getElementById("specimenList");
    const q = document.getElementById("search").value.toLowerCase();
    const stage = document.getElementById("stageFilter").value;

    list.innerHTML = "";

    specimens.forEach((s,i) => {

        const visible =
            s.title.toLowerCase().includes(q) &&
            (stage === "all" || s.stage === stage);

        if (!visible) return;

        const btn = document.createElement("button");

        btn.className = "specimen-tab" +
            (i === currentIndex ? " active" : "");

        btn.textContent = (i+1) + ". " + s.title;

        btn.onclick = () => {
            currentIndex = i;
            renderList();
            renderSpecimen();
        };

        list.appendChild(btn);
    });

    if (!list.children.length) {
        list.innerHTML = "<p>Препарат табылмады.</p>";
    }
}


// 3. ҚҰРЫЛЫМДАР ТУРАЛЫ АҚПАРАТ

function showInfo(index) {

    const s = specimens[currentIndex];
    const part = s.parts[index];

    document.getElementById("info").innerHTML =
        "<h3>" + (index+1) + ". " + part[0] + "</h3>" +
        "<p>" + part[1] + "</p>";

    document.querySelectorAll(".pin").forEach((p,i) => {
        p.classList.toggle("active",i === index);
    });
}


// 4. ПРЕПАРАТТЫ КӨРСЕТУ

function renderSpecimen() {

    const s = specimens[currentIndex];

    document.getElementById("specimenTitle").textContent = s.title;

    document.getElementById("specimenStage").textContent =
        "Даму кезеңі: " + s.stage;

    document.getElementById("description").innerHTML =
        "<b>Жалпы сипаттамасы:</b> " + s.description;

    const stage = document.getElementById("imageStage");

    stage.querySelectorAll(".pin").forEach(x => x.remove());

    const old = stage.querySelector("img");
    if (old) old.remove();

    const placeholder = document.getElementById("placeholder");

    const img = document.createElement("img");

    img.src = s.image;
    img.alt = s.title;

    img.onload = () => {
        placeholder.style.display = "none";
    };

    img.onerror = () => {
        placeholder.style.display = "block";
        placeholder.innerHTML =
            "<strong>Фото табылмады</strong><br>" +
            "Images папкасындағы файл атауын тексеріңіз.";
    };

    stage.appendChild(img);

    s.parts.forEach((part,i) => {

        const pin = document.createElement("button");

        pin.className = "pin";
        pin.textContent = i+1;

        pin.style.left = part[2] + "%";
        pin.style.top = part[3] + "%";

        pin.title = part[0];
        pin.onclick = () => showInfo(i);

        stage.appendChild(pin);
    });

    document.getElementById("info").innerHTML =
        "<h3>Нөмірді таңдаңыз</h3>" +
        "<p>Суреттегі нөмірді басып, құрылым туралы ақпаратты оқыңыз.</p>";

    
}

// 7. ТЕСТТІҢ ЖҰМЫС ІСТЕУІ

let quizIndex = 0;
let quizScore = 0;
let selectedAnswer = null;
let answered = false;

function renderQuestion() {

    const q = quizQuestions[quizIndex];

    document.getElementById("quizContent").style.display = "block";
    document.getElementById("quizResult").style.display = "none";

    document.getElementById("questionNumber").textContent =
        (quizIndex+1) + "-сұрақ / " + quizQuestions.length;

    document.getElementById("score").textContent =
        "Ұпай: " + quizScore;

    document.getElementById("progressFill").style.width =
        ((quizIndex / quizQuestions.length) * 100) + "%";

    document.getElementById("questionText").textContent =
        q.question;

    const answers = document.getElementById("answers");

    answers.innerHTML = "";

    q.options.forEach((option,index) => {

        const btn = document.createElement("button");

        btn.className = "answer";
        btn.textContent =
            String.fromCharCode(65+index) + ". " + option;

        btn.onclick = () => selectAnswer(index);

        answers.appendChild(btn);
    });

    selectedAnswer = null;
    answered = false;

    document.getElementById("feedback").className = "feedback";
    document.getElementById("feedback").style.display = "none";

    document.getElementById("checkBtn").style.display = "inline-block";
    document.getElementById("checkBtn").disabled = true;

    document.getElementById("nextBtn").style.display = "none";
}


// 8. ЖАУАПТЫ ТАҢДАУ


function selectAnswer(index) {

    if (answered) return;

    selectedAnswer = index;

    const buttons = document.querySelectorAll(".answer");

    buttons.forEach((btn,i) => {
        btn.classList.toggle("selected",i === index);
    });

    document.getElementById("checkBtn").disabled = false;
}


// 9. ЖАУАПТЫ ТЕКСЕРУ

function checkAnswer() {

    if (selectedAnswer === null || answered) return;

    answered = true;

    const q = quizQuestions[quizIndex];

    const buttons = document.querySelectorAll(".answer");

    const feedback = document.getElementById("feedback");

    if (selectedAnswer === q.correct) {

        quizScore++;

        feedback.className = "feedback correct-feedback";

        feedback.innerHTML =
            "<strong>Дұрыс жауап!</strong><br>" +
            q.explanation;

    } else {

        feedback.className = "feedback wrong-feedback";

        feedback.innerHTML =
            "<strong>Қате жауап!</strong><br>" +
            "Дұрыс жауап: " + q.options[q.correct] +
            "<br>" + q.explanation;
    }

    feedback.style.display = "block";

    buttons.forEach((btn,i) => {

        btn.disabled = true;

        if (i === q.correct) {
            btn.classList.add("correct");
        }

        if (i === selectedAnswer && selectedAnswer !== q.correct) {
            btn.classList.add("wrong");
        }

    });

    document.getElementById("score").textContent =
        "Ұпай: " + quizScore;

    document.getElementById("checkBtn").style.display = "none";

    document.getElementById("nextBtn").style.display = "inline-block";

    if (quizIndex === quizQuestions.length-1) {
        document.getElementById("nextBtn").textContent =
            "Нәтижені көру";
    } else {
        document.getElementById("nextBtn").textContent =
            "Келесі сұрақ";
    }
}


// 10. КЕЛЕСІ СҰРАҚ

function nextQuestion() {

    if (!answered) return;

    if (quizIndex < quizQuestions.length-1) {

        quizIndex++;
        renderQuestion();

    } else {

        showResult();

    }
}


// 11.  НӘТИЖЕ

function showResult() {

    document.getElementById("quizContent").style.display = "none";

    document.getElementById("quizResult").style.display = "block";

    document.getElementById("progressFill").style.width = "100%";

    document.getElementById("finalScore").textContent =
        quizScore + " / " + quizQuestions.length;

    let message = "";

    const percent = (quizScore / quizQuestions.length) * 100;

    if (percent >= 90) {
        message = "Өте жақсы! Тақырыпты өте жақсы меңгергенсің.";
    } else if (percent >= 70) {
        message = "Жақсы нәтиже! Біліміңді тағы да бекіте түс.";
    } else if (percent >= 50) {
        message = "Қанағаттанарлық. Кейбір тақырыптарды қайталау қажет.";
    } else {
        message = "Тақырыпты қайта қарап, тестті тағы бір рет орында.";
    }

    document.getElementById("resultMessage").textContent = message;
}


// 12. ТЕСТТІ ҚАЙТА БАСТАУ

function restartQuiz() {

    quizIndex = 0;
    quizScore = 0;
    selectedAnswer = null;
    answered = false;

    renderQuestion();

}
